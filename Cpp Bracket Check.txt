	
#include <iostream>
#include <stack>
#include <string>
#include <unordered_map>

using namespace std;

bool isValid(string s) {
    stack<char> st;
    // Map to easily match closing brackets to opening ones
    unordered_map<char, char> bracketMap = {
        {')', '('},
        {'}', '{'},
        {']', '['}
    };

    for (char c : s) {
        // If the character is a closing bracket
        if (bracketMap.count(c)) {
            // If stack is empty or top doesn't match the required opening bracket
            if (st.empty() || st.top() != bracketMap[c]) {
                return false;
            }
            st.pop(); // Found a match, remove from stack
        } else {
            // It's an opening bracket, push to stack
            st.push(c);
        }
    }

    // If the stack is empty, all brackets were matched
    return st.empty();
}

int main() {
    string s;
    if (!(cin >> s)) return 0;

    if (isValid(s)) {
        cout << "true" << endl;
    } else {
        cout << "false" << endl;
    }

    return 0;
}
